# Build EVTA bundle executables for the ZCU102 without the NEST-C/Glow top-level CMake.
#
# vtalib (vta_runtime + VTABundle) is added once via add_subdirectory, so every bundle target in this
# build directory shares it; adding a bundle only compiles that bundle's <name>.cpp.
# Macros/flags come from vtalib's own CMakeLists (directory-scoped add_definitions), which we read
# back and apply to the bundle targets so bundle and runtime always see the same hw_spec.
#
# usage:
#   mkdir -p build && cd build
#   cmake ..                          # clang-8 by default
#   make -j2 vtaConv3TestBundle        # binary lands in ../hw/Conv3Test/ (or: make hw_bundles)
#   cd ../hw && sudo ./run_all.sh Conv3Test
cmake_minimum_required(VERSION 3.10)
# default to clang-8 (this board's c++ is g++ 7.3; the flags below are clang-specific); -D overrides
if(NOT CMAKE_CXX_COMPILER)
  set(CMAKE_CXX_COMPILER clang++-8)
endif()
if(NOT CMAKE_C_COMPILER)
  set(CMAKE_C_COMPILER clang-8)
endif()
project(vtabundles_zcu102 CXX)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(NESTC_EVTA_RUN_WITH_GENERIC_BUNDLE ON CACHE BOOL "" FORCE)

#### JO added: bundle set = which bitstream the build targets.
####   single      (default) hw/            vta-hw-zcu102-single (or the etri zcu102_1x16_i8w8a32_16_16_19_18): buffers 16/19/18
####   multievtax2           hw_multievtax2/ vta-hw-zcu102-multi2 (or the etri zcu102_2_1x16_*): NESTC_EVTA_MULTI -> buffers 15/18/17
####   multievtax4           hw_multievtax4/ vta-hw-zcu102-multi4 (4 EVTAs, same buffers; -DVTALIB=.../vtalib-zcu102-multi4)
#### Use one build dir per set, e.g. mkdir build_multievtax2 && cd build_multievtax2 && cmake -DBUNDLE_SET=multievtax2 ..
set(BUNDLE_SET single CACHE STRING "bundle set: single, multievtax2 or multievtax4")
if(BUNDLE_SET STREQUAL "single")
  set(NESTC_EVTA_MULTI OFF CACHE BOOL "" FORCE)
elseif(BUNDLE_SET STREQUAL "multievtax2")
  set(NESTC_EVTA_MULTI ON CACHE BOOL "" FORCE)
  # Every command handle (one per EVTA) takes 2 x VTA_MAX_XFER of CMA for its uop and instruction queues
  # (vta/vtalib/include/zcu102/vta/driver.h, default 1<<25 = 32 MB -> 64 MB per handle); the board has 128 MB
  # of CMA, so two handles would take all of it. 8 MB queues: 2 x 16 MB + 2 Resnet18 copies (~13 MB weights
  # each) fit. A full instruction queue auto-syncs, so smaller queues only mean more frequent syncs.
  #### JO changed (2026-10-26): vtalib's NESTC_EVTA_MAX_XFER (its multi-EVTA default is 4 MB)
  #add_definitions(-DVTA_MAX_XFER=8388608)
  set(NESTC_EVTA_MAX_XFER 8388608 CACHE STRING "" FORCE)
elseif(BUNDLE_SET STREQUAL "multievtax4")
  set(NESTC_EVTA_MULTI ON CACHE BOOL "" FORCE)
  # four command handles: 4 MB queues -> 4 x 8 MB of CMA, and four Resnet18 copies
  #add_definitions(-DVTA_MAX_XFER=4194304)
  set(NESTC_EVTA_MAX_XFER 4194304 CACHE STRING "" FORCE)
else()
  message(FATAL_ERROR "unknown BUNDLE_SET ${BUNDLE_SET}")
endif()
set(NESTC_USE_VTASIM OFF)

# add_bundle(<dir> <target> <src>...): build hw/<dir>/<src>... into hw/<dir>/<target>.
# The sources (bundle .cpp/.o, main, *.h, *.bin) are generated by the simulator build
# (Debug2_remote/build_sim_bundles.sh) and collected into hw/<dir> by collect_hw_bundles.py.
# The executable is written next to them, so hw/run_all.sh and the per-bundle data files line up.
#### JO changed: hw dir of the bundle set
#set(HW_DIR ${CMAKE_CURRENT_SOURCE_DIR}/hw)

  set(HW_DIR ${CMAKE_CURRENT_SOURCE_DIR}/hw)
else()
  set(HW_DIR ${CMAKE_CURRENT_SOURCE_DIR}/hw_${BUNDLE_SET})
endif()
# e.g. -DBUNDLE_EXE_SUFFIX=_prof keeps a profiling build from overwriting the normal hw/<dir>/<exe>
set(BUNDLE_EXE_SUFFIX "" CACHE STRING "suffix appended to every bundle executable name")
# -DBUNDLE_TIMED=ON compiles a per-statement timed copy of each bundle .cpp (timed_bundle.py);
# use with its own build dir and -DBUNDLE_EXE_SUFFIX=_timed. Mains are compiled unchanged.
option(BUNDLE_TIMED "time every statement of the generated bundle functions" OFF)
find_package(PythonInterp 3 REQUIRED)
add_custom_target(hw_bundles)
function(add_bundle dir target)
  set(srcs)
  #### JO added (2026-10-26): several timed .cpp in one executable (multi-EVTA copies, VTA partitions) write
  #### timing_<name>.csv each instead of all overwriting timing.csv
  set(ntimed 0)
  foreach(f ${ARGN})
    if(f MATCHES "\\.cpp$" AND NOT f MATCHES "([Mm]ain)[^/]*\\.cpp$")
      math(EXPR ntimed "${ntimed} + 1")
    endif()
  endforeach()
  foreach(f ${ARGN})
    if(BUNDLE_TIMED AND f MATCHES "\\.cpp$" AND NOT f MATCHES "([Mm]ain)[^/]*\\.cpp$")
      #### JO changed (2026-10-26): per target (MultiEVTAResnet18Test: mxnet_exported_resnet18.cpp is in two targets)
      #set(timed ${CMAKE_CURRENT_BINARY_DIR}/timed/${dir}/${f})
      set(timed ${CMAKE_CURRENT_BINARY_DIR}/timed/${dir}/${target}/${f})
      set(csv timing.csv)
      if(ntimed GREATER 1)
        get_filename_component(stem ${f} NAME_WE)
        set(csv timing_${stem}.csv)
      endif()
      add_custom_command(OUTPUT ${timed}
        COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/timed/${dir}/${target}
        #COMMAND ${PYTHON_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/timed_bundle.py ${HW_DIR}/${dir}/${f} ${timed}
        COMMAND ${PYTHON_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/timed_bundle.py ${HW_DIR}/${dir}/${f} ${timed} ${csv}
        DEPENDS ${HW_DIR}/${dir}/${f} ${CMAKE_CURRENT_SOURCE_DIR}/timed_bundle.py)
      list(APPEND srcs ${timed})
    else()
      list(APPEND srcs ${HW_DIR}/${dir}/${f})
    endif()
  endforeach()
  add_executable(${target} EXCLUDE_FROM_ALL ${srcs})
  target_compile_definitions(${target} PRIVATE ${VTA_DEFS})
  target_compile_options(${target} PRIVATE ${VTA_OPTS})
  target_include_directories(${target} PRIVATE
    ${HW_DIR}/${dir} ${NESTC_ROOT} ${VTA_INCS} ${VTALIB}/include/Bundle)
  target_link_libraries(${target} VTABundle png tinfo dl z)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${HW_DIR}/${dir}
    OUTPUT_NAME ${target}${BUNDLE_EXE_SUFFIX})
  add_dependencies(hw_bundles ${target})
endfunction()

# register every hw/<dir> that has <X>Bundle.cpp + <X>BundleMain.cpp
file(GLOB bundle_mains RELATIVE ${HW_DIR} ${HW_DIR}/*/*BundleMain.cpp)
foreach(m ${bundle_mains})
  get_filename_component(dir ${m} DIRECTORY)
  get_filename_component(file ${m} NAME_WE)
  string(REGEX REPLACE "Main$" "" target ${file})
  if(EXISTS ${HW_DIR}/${dir}/${target}.cpp)
    add_bundle(${dir} ${target} ${target}Main.cpp ${target}.cpp)
  endif()
endforeach()

# bundles with other file names (sources as in nest-compiler/vta/bundles/<dir>/CMakeLists.txt)
function(add_bundle_if_present dir target)
  if(EXISTS ${HW_DIR}/${dir}/${ARGV2})
    add_bundle(${dir} ${target} ${ARGN})
  endif()
endfunction()
